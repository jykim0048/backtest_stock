	$('a.btn.sitemap.open').bind("click focus",function(e){
		e.preventDefault();		
		if(cnt==0){
			$(this).addClass('hide');
			cnt=1;
			setTimeout(function(){
				$('section.sitemap').slideToggle(400);
			},200);
		}else if(cnt==1){
			setTimeout(function(){
				$('section.sitemap').slideToggle(400);				
			},200);			
			setTimeout(function(){
				$('a.btn.sitemap.open').removeClass('hide');
			},400);
			
			cnt=0;
		}
	});
	$('a.btn_sitemap-close').bind("click",function(e){
		e.preventDefault();
		setTimeout(function(){
			$('section.sitemap').slideToggle(400,function(){
				cnt=0;
				$('a.btn.sitemap.open').removeClass('hide');
			});
		},100);
	});
	
	$('section.sitemap a.first').bind("mouseover focus",function(){
		$('section.sitemap article').removeClass("active");
		$(this).parent().addClass("active");
		$(this).parent().parent().parent().addClass("active");
	});
	$('section.sitemap a.first').bind("mouseout focusout",function(){
		$(this).parent().removeClass("active");
	});

	$('section.sitemap span , section.sitemap input[type=checkbox]').bind("mouseover focus",function(){

function fnInMyMenu(){
	var cookies = GetCookie('krxMenu');
	var cookieValue = "";
	if(cookies == null ){
		cookieValue = cookieValue + "ULDDST00000,오늘의공시/";
		cookieValue = cookieValue + "ULDDST00100,회사별검색/";
		cookieValue = cookieValue + "ULDDST00300,상세검색/";
		cookieValue = cookieValue + "ULDDST00200,통합검색/";
		cookieValue = cookieValue + "ULDDST71000,예비심사기업/";
		SetCookie('krxMenu',cookieValue ,'7');
	} else {

	}

	fnUpdateInMyMenu();

function fnSetCookieMyMenu(){
	var cookieValue = "";
	$('input:checkbox[name="checkboxMenu"]:checked').each(function(){
		var menuName = $("label[for="+this.id+"]").text().replace("/","");
		cookieValue += $(this).val()+","+menuName+"/";
	});
	SetCookie('krxMenu',cookieValue ,'7');
	fnUpdateInMyMenu();
}

function fnUpdateInMyMenu(){
	var cookies = GetCookie('krxMenu');
	if(cookies != null ){
		$("#myMenuTop").empty();
		var items = cookies.split("/");
		for(var i in items){
			var item = items[i].split(",");
			var menuId	 = item[0];
			var menuName = item[1];
			if(undefined != menuName){
				var url = fnGetInMyMenuUrl(menuId);
				// My Menu 설정을 많이 해서 스크롤이 생겼을 경우 길이가 8자 이상인 메뉴명이 포함되어 있으면 메뉴 부분이 깨지는 현상이 있어서 ..으로 처리함.
				if(menuId == "ULDDST00400") {		// 'ETF ETN ELW 상장형수익증권' 메뉴명은 영어로 구성되어 있어서 6자리 보다 11자리를 짜름. 
					menuName = menuName.substring(0,11) + "..";
				}
				else {
					if(menuName.length >= '8') {
						menuName = menuName.substring(0,6) + "..";
					}
				}
				
				// 신규 추가 된 밸류업 보고서의 링크는 새창에서 열려야 하기 때문에 target 추가함. 2025.08.29
				if(menuId == "ULDDST80X00") {
					$("#myMenuTop").append("<li><a href='"+url+"' target='_brank'>"+menuName+"</a></li>");
				} 
				else {
					$("#myMenuTop").append("<li><a href='"+url+"'>"+menuName+"</a></li>");
				}
				
			}

			$('input:checkbox[name="checkboxMenu"]').each(function(){
				if($(this).val() == menuId)
					this.checked = true;
			});
		}
	}
}

document.onkeypress = function(e){
	var result = "";
	if(typeof(e)!='undefined'){
		result = e.which;
	}else{
		result = event.keyCode;
	}
	keyEvent(result);
}

function keyEvent(e){
	if(document.all){//IE
		if(window.event.keyCode==13){
            if(fnAKCKwdSearchCodeType(eval(document.searchForm.searchCorpNameTmp))) {
            	fnSearchCorpName();
            }
		}
	}else{
		if(e==13){
            if(fnAKCKwdSearchCodeType(eval(document.searchForm.searchCorpNameTmp))) {
            	fnSearchCorpName();
            }
		}
	}
}

function fnViewTab(tabDiv) {

	$(document).find("li[id=menuType]").each(function (index){
		if($(this).hasClass("active")) {
			$(this).removeClass("active");

		}
	});

	fnSearchInit();
}
